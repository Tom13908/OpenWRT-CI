#================================================================
# CMIOT AX18 精简固件配置（qualcommax / ipq60xx）
# 配合 VIKINGYFY/immortalwrt 源码（自带 NSS）
# 用法：放入 OpenWRT-CI 仓库的 Config/ 目录，并在 QCA-ALL.yml
#       matrix 的 CONFIG 列表中改为 CMIOT-AX18
#================================================================

#----- 设备平台 -----
CONFIG_TARGET_qualcommax=y
CONFIG_TARGET_qualcommax_ipq60xx=y

#----- 仅保留 CMIOT AX18 一个设备（精简，跳过其它 17 个 ipq60xx 设备）-----
CONFIG_TARGET_DEVICE_qualcommax_ipq60xx_DEVICE_cmiot_ax18=y

#----- ZRAM 内存压缩 -----
CONFIG_PACKAGE_kmod-zram=y
CONFIG_PACKAGE_zram-swap=y

#----- iStore 应用商店（编译期集成）-----
# deps 会自动带 luci-lib-taskd / taskd / luci-lib-xterm，无需 luci-lib-ipkg
CONFIG_PACKAGE_luci-app-store=y

#----- 常用维护插件（精简）-----
CONFIG_PACKAGE_luci-app-upnp=y
CONFIG_PACKAGE_luci-app-commands=y
CONFIG_PACKAGE_luci-app-package-manager=y
